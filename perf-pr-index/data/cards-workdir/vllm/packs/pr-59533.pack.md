## Overview

**Projection fusion.** Merge Qwen4Exp QSA’s QKVG and replicated indexer Q/K projections into **one GEMM**. LoRA keeps separate projections with standard linear dispatch.

**GEMM dispatch.** LL-GEMM for merged GEMM

- SM103 (GB300): TP4 shape, done by me
- SM100 (B200): TP4 contributed by @namgyu-youn
- SM90 (H200): TP4 contributed tuning for M=1/2/4
- SM121 (GB10): TP1 contributed by @stecasta
- Also prune obsolete merged-HC and separate-QSA plans;

**Prepare kernel.** Support strided merged outputs and widen row-offset arithmetic to i64, with **no noticeable performance difference** in GB300 checks (≤1% in repeat measurements).

## Validation

### Microbenchmark

```python
import ast
import dataclasses
import json
import statistics
import subprocess
from pathlib import Path
from unittest.mock import patch

import torch
from flashinfer.testing import bench_gpu_time_with_cupti
from vllm.models.qwen4_exp.nvidia import low_latency_gemm as dispatch

M_VALUES = (1, 4, 16, 64, 256, 1024, 4096, 16384)
ROUNDS = 3
REPEAT_ITERS = 50
BASELINE_COMMIT = "9e6550bd058560bec5690b6c06474adf048a9b85"


def baseline_plans():
    source = subprocess.check_output([
        "git", "show",
        f"{BASELINE_COMMIT}:vllm/models/qwen4_exp/nvidia/low_latency_gemm.py",
    ], text=True)
    node = next(node for node in ast.parse(source).body
                if isinstance(node, ast.AnnAssign)
                and node.target.id == "QWEN4_EXP_GEMM_PLANS")
    namespace = {"SkinnyGemmConfig": dispatch.SkinnyGemmConfig}
    exec(compile(ast.Module(body=[node], type_ignores=[]), "baseline-plans", "exec"), namespace)
    return namespace["QWEN4_EXP_GEMM_PLANS"]


def timing(fn, cold):
    samples = bench_gpu_time_with_cupti(
        fn, dry_run_iters=5, repeat_iters=REPEAT_ITERS,
        use_cuda_graph=True, cold_l2_cache=cold,
    )
    return statistics.median(samples) * 1000


def main():
    torch.manual_seed(42)
    weight = torch.randn(4224, 2560, device='cuda', dtype=torch.bfloat16) * 0.02
    qkv_weight, index_weight = weight.split((3584, 640), dim=0)
    real_skinny = dispatch.shape_dynamic_skinny_gemm
    records = []
    def spy(x, w, config):
        records.append((tuple(w.shape), dataclasses.asdict(config)))
        return real_skinny(x, w, config)
    results = {
        'gpu': torch.cuda.get_device_name(), 'torch': torch.__version__,
        'baseline_commit': BASELINE_COMMIT,
        'candidate_commit': subprocess.check_output(['git', 'rev-parse', 'HEAD'], text=True).strip(),
        'dtype': 'bf16', 'tp': 4, 'k': 2560, 'separate_n': [3584, 640],
        'merged_n': 4224, 'm_values': M_VALUES, 'rounds': ROUNDS,
        'iterations_per_round': REPEAT_ITERS, 'cuda_graph': True, 'results': [],
    }
    old_plans = baseline_plans()
    capability = dispatch.current_platform.get_device_capability()
    merged_plans = dispatch.QWEN4_EXP_GEMM_PLANS_BY_CAPABILITY.get(capability, {})
    op = torch.ops.vllm.qwen4_exp_low_latency_gemm
    for m in M_VALUES:
        x = torch.randn(m, 2560, device='cuda', dtype=torch.bfloat16)
        def separate():
            with patch.dict(dispatch.QWEN4_EXP_GEMM_PLANS_BY_CAPABILITY, {capability: old_plans}):
                return op(x, qkv_weight), op(x, index_weight)
        def merged():
            return op(x, weight)
        arms = {'separate': separate, 'merged': merged}
        row = {'m': m, 'dispatch': {}}
        for name, fn in arms.items():
            records.clear()
            with patch.object(dispatch, 'shape_dynamic_skinny_gemm', side_effect=spy):
                output = fn()
            row['dispatch'][name] = list(records)
            sizes = (3584, 640) if name == 'separate' else (4224,)
            plans = old_plans if name == "separate" else merged_plans
            expected = sum(m in plans.get((n, 2560), {}) for n in sizes)
            assert len(records) == expected, (m, name, expected, records)
            outputs = output if isinstance(output, tuple) else output.split((3584, 640), dim=1)
            for actual, w in zip(outputs, (qkv_weight, index_weight)):
                reference = torch.nn.functional.linear(x.float(), w.float()).bfloat16()
                torch.testing.assert_close(actual, reference, rtol=2e-2, atol=1e-2)
            for _ in range(10):
                fn()
        torch.cuda.synchronize()
        for cold in (True,):
            mode = 'cold' if cold else 'hot'
            times = {name: [] for name in arms}
            for run in range(ROUNDS):
                order = ('separate', 'merged') if run % 2 == 0 else ('merged', 'separate')
                for name in order:
                    times[name].append(timing(arms[name], cold))
            row[mode] = {
                name: {'median_us': statistics.median(ts), 'run_medians_us': ts,
                       'tflops': 2*m*2560*4224 / (statistics.median(ts)*1e6)}
                for name, ts in times.items()
            }
            row[mode]['speedup'] = row[mode]['separate']['median_us'] / row[mode]['merged']['median_us']
        results['results'].append(row)
        Path('workspace/qsa_projection_dispatch_results.json').write_text(json.dumps(results, indent=2)+'\n')
        print(json.dumps(row), flush=True)


if __name__ == '__main__':
    main()
```
