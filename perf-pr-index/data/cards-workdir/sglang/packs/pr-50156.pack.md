## Motivation

These are runtime optimizations for serving with MTP speculative decoding on Blackwell / Rubin, aimed at agentic multi-turn workloads. They sit **outside the per-step model forward**:
- speculative-decoding verify and post-verify work (verify-tree sampling, the draft top-1, the post-verify graphs and their pool, the verify commit's state rollbacks, the compact verify support's top-p bound);
- CUDA-graph input staging;
- HiCache and the Rust TreeCore (KV and Mamba state loads, backups, host eviction order, tracking slots, write-back demotion);
- host-side batch preparation (text-only mrope rows) and the PLE host table.

Every optimization is opt-in. It runs only with the runtime master `SGLANG_ENABLE_NV_MEGA_RUNTIME_OPTIMIZATION` plus its own child switch (see `environ.py` and `test_environ.py`). With the defaults, behavior is unchanged.

### Scope and related work

- **Smaller than the previous revision.** Each commit of the previous revision was measured on its own (switch off against the full series, on the workloads named in the Gain lines). - **Base:** this PR is standalone on `main`.

+SGL_DEVICE void copy_span(const char* src, char* dst, const int64_t len) {
+  for (int64_t off0 = static_cast<int64_t>(threadIdx.x) * kBytesPerThreadPerStep; off0 < len;
+       off0 += kWordsInFlight * kBytesPerBlockPerStep) {
+    uint4 v[kWordsInFlight];
+#pragma unroll
+    for (int k = 0; k < kWordsInFlight; ++k) {
+      const int64_t off = off0 + k * kBytesPerBlockPerStep;
+      if (off < len) v[k] = device::details::load_nc(reinterpret_cast<const uint4*>(src + off));
+    }
+#pragma unroll
+    for (int k = 0; k < kWordsInFlight; ++k) {
+      const int64_t off = off0 + k * kBytesPerBlockPerStep;
+      if (off < len) device::details::store_nc(reinterpret_cast<uint4*>(dst + off), v[k]);

+__global__ void __launch_bounds__(kTreeThreads) pv_tree_final_kernel(
+    IdType* predicts,
+    const DType* uniform_samples_for_final_sampling,
+    const DType* target_probs,
+    const DType* draft_probs,

+  ::sglang::device::PDLWaitPrimary<true>();

+  const uint32_t num_accepted_tokens = static_cast<uint32_t>(meta[bx * kMetaWords + 1]);

+    const DType u = coin * sum_relu_q_minus_p;
+    DType aggregate(0);
+    int chunk = -1;
+    int last_valid_id = -1;
+    for (uint32_t i = 0; i < nchunks; ++i) {
+      const DType aggregate_local = thresholded_sums[i];
+      if (aggregate + aggregate_local > u) {
+        chunk = static_cast<int>(i);
+        break;
+      }
+      const int max_valid_index = max_valid[i];

+      }
+      aggregate += aggregate_local;
+    }
+    temp_storage.sampled_id = d;
+    temp_storage.last_valid_id = last_valid_id;
+    s_u = u;
+    s_aggregate = aggregate;

+        num_accepted_tokens != num_speculative_tokens - 1,
+        static_cast<uint32_t>(chunk),
+        d);

+    int last_valid_index = temp_storage.last_valid_id;  // the earlier chunks' maximum
+    DeviceSamplingFromProb<VEC_SIZE, kTreeThreads, SCAN_ALGO, REDUCE_ALGO, DETERMINISTIC>(
+        static_cast<uint32_t>(chunk),
+        d,

+        s_u,
+        relu_q_minus_p_vec,
+        aggregate_relu_q_minus_p,
+        last_valid_index,
+        &temp_storage);
+  }
+  __syncthreads();

+  if (sampled_id == d) {

+    } else {

+    }
+  }

+def _compact_commit_tile(

+            other=0.0,
