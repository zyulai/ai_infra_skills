+//! Multimodal model support that lives in this repository until
+//! `llm-multimodal` ships it.

+//! Both tables are consulted before the `llm-multimodal` registry. Once
+//! `llm-multimodal` supports a model, delete its entry in the same change that

+mod step3_vl;

+/// Specs for models that `llm-multimodal` does not have yet, one model per
+/// submodule.
+static SPECS: &[&dyn ModelProcessorSpec] = &[
+    &step3_vl::Step3VLVisionSpec, // Step3-VL, Step5-VL
+];
+
+/// `(model_type, target model_type)` pairs for checkpoints that reuse another
+/// family's `llm-multimodal` spec unchanged. The target spec still sees this
+/// model's own metadata.
+const MODEL_TYPE_ALIASES: &[(&str, &str)] = &[];
+
+/// Resolve the model processor spec for one loaded model: a local spec first,
+/// then the `llm-multimodal` registry, through a model type alias if any.
+pub(super) fn lookup_spec(metadata: &ModelMetadata<'_>) -> Option<&'static dyn ModelProcessorSpec> {
+    static LLM_MULTIMODAL_REGISTRY: LazyLock<ModelRegistry> = LazyLock::new(ModelRegistry::new);
+
+    if let Some(spec) = SPECS.iter().copied().find(|spec| spec.matches(metadata)) {
+        return Some(spec);
+    }
+    let target = metadata.config_model_type().and_then(|model_type| {
+        MODEL_TYPE_ALIASES
+            .iter()
+            .find_map(|&(alias, target)| (alias == model_type).then_some(target))
+    });
+    let Some(target) = target else {
+        return LLM_MULTIMODAL_REGISTRY.lookup(metadata);
+    };
+    let mut config = metadata.config.clone();
+    config["model_type"] = target.into();
+    LLM_MULTIMODAL_REGISTRY.lookup(&ModelMetadata {
+        config: &config,
+        ..*metadata
+    })
+}

+//! Step3-VL vision model contract (also used by Step5-VL in vLLM).
+//!
+//! Mirrors vLLM's `transformers_utils/processors/step3_vl.py`: each `<im_patch>`
+//! placeholder expands to per-patch spans (`<patch_start>` + 81 embed tokens +
+//! `<patch_end>`, with `<patch_newline>` after row-final patches) followed by the
+//! global-view span (`<im_start>` + 169 embed tokens + `<im_end>`). The per-image
+//! patch counts and newline flags come from the preprocessor's `num_patches` and
+//! `patch_newline_mask` model-specific outputs.
+
+use std::collections::HashMap;
+
+use serde_json::{Value, json};
+
+use llm_multimodal::registry::{ModelRegistryError, RegistryResult};
+use llm_multimodal::{
+    FieldLayout, Modality, ModelMetadata, ModelProcessorSpec, ModelSpecificValue,
+    PreProcessorConfig, PreprocessedEncoderInputs, PromptReplacement, TokenId, VisionPreProcessor,
+};
+
+use self::processor::Step3VLProcessor;

+mod processor;

+pub(super) struct Step3VLVisionSpec;

+impl Step3VLVisionSpec {
+    const IMAGE_TOKEN: &'static str = "<im_patch>";
+    const IMAGE_START_TOKEN: &'static str = "<im_start>";
+    const IMAGE_END_TOKEN: &'static str = "<im_end>";
+    const PATCH_START_TOKEN: &'static str = "<patch_start>";
+    const PATCH_END_TOKEN: &'static str = "<patch_end>";
+    const PATCH_NEWLINE_TOKEN: &'static str = "<patch_newline>";
+
+    // Defaults from Step3VLImageProcessor; the processor reads overrides from
+    // PreProcessorConfig extras, and `processor_kwargs` exposes these knobs.
+    const NUM_IMAGE_FEATURE_SIZE: usize = 169;
+    const NUM_PATCH_FEATURE_SIZE: usize = 81;
+
+    /// Per-image patch counts from the preprocessor output. Missing or
+    /// malformed data falls back to zero patches per image (global view only).
+    fn extract_num_patches(preprocessed: &PreprocessedEncoderInputs) -> Vec<usize> {
+        if let Some(ModelSpecificValue::IntTensor { data, .. }) =
+            preprocessed.model_specific.get("num_patches")
+        {
+            return data.iter().map(|&n| n.max(0) as usize).collect();
+        }
+        vec![0; preprocessed.feature_token_counts.len()]
+    }
+
+    /// Flat per-patch newline flags from the preprocessor output.
+    fn extract_newline_mask(preprocessed: &PreprocessedEncoderInputs) -> Vec<u32> {
+        if let Some(ModelSpecificValue::UintTensor { data, .. }) =
+            preprocessed.model_specific.get("patch_newline_mask")
+        {
+            return data.clone();
+        }
+        Vec::new()
+    }
+}

+impl ModelProcessorSpec for Step3VLVisionSpec {
+    fn name(&self) -> &'static str {
+        "step3_vl"
+    }
+
+    fn matches(&self, metadata: &ModelMetadata) -> bool {
+        if metadata
+            .config_model_type()
+            .is_some_and(|mt| mt == "step5_vl" || mt == "step3_vl")
+        {
+            return true;
+        }
+        let id = metadata.model_id.to_ascii_lowercase();
+        id.contains("step5-vl")
+            || id.contains("step5_vl")
+            || id.contains("step3-vl")
+            || id.contains("step3_vl")
+    }
+
+    fn placeholder_token(&self, _metadata: &ModelMetadata) -> RegistryResult<String> {
+        Ok(Self::IMAGE_TOKEN.to_string())
+    }
+
+    fn placeholder_token_id(&self, metadata: &ModelMetadata) -> RegistryResult<TokenId> {
+        metadata.token_id(Self::IMAGE_TOKEN)
+    }
+
+    fn modality_limits(
+        &self,
+        _metadata: &ModelMetadata,
+    ) -> RegistryResult<HashMap<Modality, usize>> {
+        Ok(HashMap::from([(Modality::Image, usize::MAX)]))
+    }

+    fn processor_kwargs(&self, _metadata: &ModelMetadata) -> RegistryResult<Value> {
+        Ok(json!({
+            "image_size": 728,
+            "patch_size": 504,
+            "num_image_feature_size": Self::NUM_IMAGE_FEATURE_SIZE,
+            "num_patch_feature_size": Self::NUM_PATCH_FEATURE_SIZE,
+            "enable_patch": true,
+        }))
+    }

+    fn vision_processor(
+        &self,
+        _metadata: &ModelMetadata,
+        config: &PreProcessorConfig,
+        modality: Modality,
+    ) -> RegistryResult<Box<dyn VisionPreProcessor>> {
+        match modality {
+            Modality::Image => Ok(Box::new(Step3VLProcessor::from_preprocessor_config(config))),
+            _ => Err(ModelRegistryError::UnsupportedModality {
+                spec: self.name(),
+                modality,
+            }),
+        }
+    }

+    fn prompt_replacements(
+        &self,
+        metadata: &ModelMetadata,
+        preprocessed: &PreprocessedEncoderInputs,
+    ) -> RegistryResult<Vec<PromptReplacement>> {
+        let image_token_id = metadata.token_id(Self::IMAGE_TOKEN)?;
+        let im_start_id = metadata.token_id(Self::IMAGE_START_TOKEN)?;
+        let im_end_id = metadata.token_id(Self::IMAGE_END_TOKEN)?;
+        let patch_start_id = metadata.token_id(Self::PATCH_START_TOKEN)?;
+        let patch_end_id = metadata.token_id(Self::PATCH_END_TOKEN)?;
+        let patch_newline_id = metadata.token_id(Self::PATCH_NEWLINE_TOKEN)?;
+
+        let num_patches = Self::extract_num_patches(preprocessed);
+        let newline_mask = Self::extract_newline_mask(preprocessed);
+
+        let mut replacements = Vec::with_capacity(num_patches.len());
+        let mut offset = 0usize;
+        for &n in &num_patches {
+            // Mirrors Step3VLProcessor.get_image_repl_feature_ids: patch spans
+            // first, then the global-view span.
+            let mut tokens = Vec::with_capacity(
+                n * (Self::NUM_PATCH_FEATURE_SIZE + 2) + Self::NUM_IMAGE_FEATURE_SIZE + 2,
+            );
+            for p in 0..n {
+                tokens.push(patch_start_id);
+                tokens.extend(std::iter::repeat_n(
+                    image_token_id,
+                    Self::NUM_PATCH_FEATURE_SIZE,
+                ));
+                tokens.push(patch_end_id);
+                if newline_mask.get(offset + p).copied().unwrap_or(0) == 1 {
+                    tokens.push(patch_newline_id);
+                }
+            }
+            offset += n;
+
+            tokens.push(im_start_id);
+            tokens.extend(std::iter::repeat_n(
+                image_token_id,
+                Self::NUM_IMAGE_FEATURE_SIZE,
+            ));
+            tokens.push(im_end_id);
+
+            replacements.push(PromptReplacement::sequence(
+                Modality::Image,
+                Self::IMAGE_TOKEN,
+                tokens,
+            ));
+        }
+        Ok(replacements)
+    }
+
+    fn field_layouts(&self) -> HashMap<String, FieldLayout> {
+        HashMap::from([
+            ("pixel_values".to_string(), FieldLayout::Batched),
+            (
+                "patch_pixel_values".to_string(),
+                FieldLayout::flat("num_patches"),
+            ),
+            (
+                "patch_newline_mask".to_string(),
+                FieldLayout::flat("num_patches"),
+            ),
+            ("num_patches".to_string(), FieldLayout::Batched),
+        ])
+    }

+//! Step3-VL vision preprocessor.
+//!
+//! Port of vLLM's `transformers_utils/processors/step3_vl.py` (also the
+//! frontend Step5-VL reuses): a 728x728 global view plus a
+//! sliding-window grid of 504x504 patches, both CLIP-normalized, with a
+//! per-patch `<patch_newline>` mask marking row ends in reading order.
+//!
+//! Each `<im_patch>` placeholder expands to
+//! `<im_start> <im_patch>*169 <im_end>` for the global view plus, per patch,
+//! `<patch_start> <im_patch>*81 <patch_end>` and a `<patch_newline>` wherever
+//! the mask is set. The integer/truncation semantics of the reference
+//! (`int()`, `//`, float ratios) are replicated exactly, including the quirk
+//! where the crop-resize decision compares the crop size against the post-pad
+//! dimensions rather than the post-preprocess dimensions.

+const MAX_IMAGE_SIZE: u32 = 3024;
+/// Hard-coded threshold inside the reference `determine_window_size`.
+const WINDOW_LONG_SIDE: u32 = 728;
+/// Hard-coded default window inside the reference `determine_window_size`.
+const WINDOW_DEFAULT: u32 = 504;

+const DEFAULT_IMAGE_SIZE: u32 = 728;
+const DEFAULT_PATCH_SIZE: u32 = 504;
+const DEFAULT_NUM_IMAGE_FEATURE_SIZE: usize = 169;
+const DEFAULT_NUM_PATCH_FEATURE_SIZE: usize = 81;

+/// Long/short side -> sliding window size (0 disables patching).
+fn determine_window_size(long: u32, short: u32) -> u32 {
+    let ratio = long as f64 / short as f64;
+    if long < WINDOW_LONG_SIDE {
+        return if ratio > 1.5 { short } else { 0 };
+    }
+    if ratio > 4.0 {
+        short.min(WINDOW_DEFAULT)
+    } else {
+        WINDOW_DEFAULT
+    }
+}

+/// Extreme-aspect tiny images are square-padded before anything else.
+fn get_image_size_for_padding(width: u32, height: u32) -> (u32, u32) {
+    let ratio = width as f64 / height as f64;
+    if width.min(height) < 32 && !(0.25..=4.0).contains(&ratio) {
+        let size = width.max(height);
+        (size, size)
+    } else {
+        (width, height)
+    }
+}
+
+/// Cap the longer side at `MAX_IMAGE_SIZE` (Python `int()` truncation).
+fn get_image_size_for_preprocess(width: u32, height: u32) -> (u32, u32) {
+    let max_side = width.max(height);
+    if max_side > MAX_IMAGE_SIZE {
+        let scale = MAX_IMAGE_SIZE as f64 / max_side as f64;
+        (
+            (width as f64 * scale) as u32,
+            (height as f64 * scale) as u32,
+        )
+    } else {
+        (width, height)
+    }
+}

+/// Round one dimension up to a window multiple; a fractional part above 0.2
+/// adds a whole extra window.
+fn crop_dim(dim: u32, window_size: u32) -> u32 {
+    let ratio = dim as f64 / window_size as f64;
+    if ratio < 1.0 {
+        return dim;
+    }
+    let decimal = ratio - (dim / window_size) as f64;
+    let n = if decimal > 0.2 {
+        ratio as u32 + 1
+    } else {
+        ratio as u32
+    };
+    window_size * n
+}
+
+fn get_image_size_for_crop(width: u32, height: u32, window_size: u32) -> (u32, u32) {
+    (crop_dim(width, window_size), crop_dim(height, window_size))
+}

+/// Window starts along one axis; the last start is clamped so the window
+/// stays inside the extent.
+fn window_starts(extent: u32, window_size: u32) -> Vec<u32> {
+    let num = if extent <= window_size {
+        1
+    } else {
+        (((extent - window_size) as f64 / window_size as f64) + 1.0).ceil() as u32
+    };
+    let mut starts: Vec<u32> = (0..num).map(|i| window_size * i).collect();
+    if num > 1 && starts[num as usize - 1] + window_size > extent {
+        starts[num as usize - 1] = extent - window_size;
+    }
+    starts
+}
+
+/// A sliding-window box as (x, y, width, height).
+type Window = (u32, u32, u32, u32);
+
+/// Sliding windows over a (width, height) extent, y-outer x-inner order.
+/// Returns (windows, x_num, y_num).
+fn slide_window(width: u32, height: u32, window_size: u32) -> (Vec<Window>, u32, u32) {
+    let x_start = window_starts(width, window_size);
+    let y_start = window_starts(height, window_size);
+    let mut windows = Vec::with_capacity(x_start.len() * y_start.len());
+    for &y in &y_start {
+        for &x in &x_start {
+            windows.push((x, y, window_size, window_size));
+        }
+    }
+    (windows, x_start.len() as u32, y_start.len() as u32)
+}

+/// Reference `ImageOps` square pad: black canvas, paste at top-left.
+fn square_pad(img: &RgbImage) -> RgbImage {
+    let (w, h) = img.dimensions();
+    let size = w.max(h);
+    let mut padded = RgbImage::from_pixel(size, size, Rgb([0, 0, 0]));
+    image::imageops::overlay(&mut padded, img, 0, 0);
+    padded
+}

+    fn get_num_patches(&self, width: u32, height: u32) -> (usize, usize) {
+        let (width, height) = get_image_size_for_padding(width, height);
+        let (width, height) = get_image_size_for_preprocess(width, height);
+        let window_size = determine_window_size(width.max(height), width.min(height));
+        if window_size == 0 || !self.enable_patch {
+            return (0, 0);
+        }

+    /// Reference `ImagePatcher.__call__`: split one decoded image into its
+    /// global view and window patches with the newline mask.
+    fn patch_one(&self, image: &DynamicImage) -> Result<PatchedImage, TransformError> {
+        let rgb = image.to_rgb8();
+        let (mut width, mut height) = rgb.dimensions();
+
+        let rgb = match get_image_size_for_padding(width, height) {
+            (pw, ph) if (pw, ph) != (width, height) => {
+                width = pw;
+                height = ph;
+                square_pad(&rgb)
+            }
+            _ => rgb,
+        };
+
+        let (new_width, new_height) = get_image_size_for_preprocess(width, height);
+        if new_width == 0 || new_height == 0 {
+            return Err(TransformError::ShapeError(format!(
+                "step3_vl preprocess size collapsed to {new_width}x{new_height} \
+                 for input {width}x{height}"
+            )));
+        }
+        let global = resize(
+            &DynamicImage::ImageRgb8(rgb),
+            new_width,
+            new_height,
+            FilterType::Triangle,
+        )
+        .to_rgb8();
+
+        let window_size =
+            determine_window_size(new_width.max(new_height), new_width.min(new_height));
+        if window_size == 0 || !self.enable_patch {
+            return Ok(PatchedImage {
+                global,
+                patches: Vec::new(),
+                newline_mask: Vec::new(),
+            });
+        }
+
+        let (crop_width, crop_height) = get_image_size_for_crop(new_width, new_height, window_size);
+        // Reference quirk: the resize decision compares against the post-pad
+        // dimensions, not the post-preprocess ones.

+                &DynamicImage::ImageRgb8(global.clone()),
+                crop_width,
+                crop_height,
+                FilterType::Triangle,
+            )
+            .to_rgb8()
+        } else {
+            global.clone()
+        };
+
+        let (windows, x_num, _y_num) = slide_window(crop_width, crop_height, window_size);
+        let mut patches = Vec::with_capacity(windows.len());
+        let mut newline_mask = vec![0u32; windows.len()];
+        for (patch_id, &(x, y, pw, ph)) in windows.iter().enumerate() {
+            patches.push(image::imageops::crop_imm(&img_for_crop, x, y, pw, ph).to_image());

+                newline_mask[patch_id] = 1;
+            }
+        }
+        if let Some(last) = newline_mask.last_mut() {
+            *last = 0;
+        }
+
+        Ok(PatchedImage {
+            global,
+            patches,
+            newline_mask,
+        })
+    }
