## Motivation

Split #44747 into reviewable parts. This is the kernels part, position 3 of 7 in the merge order below.

## Modifications

- **Beam fused ops**, listed by class method as in Muse Spark's kernels part:
  - `elementwise.beam_attention_gate`: the per-head gate, with Beam's activation-dtype bias and sigmoid rounding.
  - `layernorm.beam_sandwich_residual_rmsnorm`: the post-norm, depth-scaled residual and next RMSNorm in one launch.
- **`communication.beam_all_reduce_norm`**: Beam's BF16 TP all-reduce fused with that norm and residual, on CustomAllReduceV2's push plane. Each row runs on one CTA, or on a 5-CTA cluster at TP8 for up to 16 rows. It needs a new `CustomAllReduceV2.supports_push` capacity check.
- **FP32 router gates:**
  - FP32 weights in main's `tiny_gemm` (N variant) and `router_gemv` (CUDA, up to 8 rows);
  - `gemm.router_gemm_bf16x3_cublas` for larger batches, with the weight split per call by `quantization.split_fp32_to_bf16x3`.

Main's existing callers are unchanged:
- BF16 tiny GEMM builds the same kernel with the same split as on main (its JIT build key gains the weight type, so it recompiles once on first use);
- the gfx950 BF16 router GEMV path is unchanged.

`gemm.router_gemv` is registered for Triton CUDA/HIP; its existing predicate retains CUDA FP32 and gfx950 BF16 gating. `split_fp32_to_bf16x3` lives in `ops.quantization.bf16x3_split`, which #45361 now imports directly. The cuBLAS wrapper keeps its internal import.

Committed coverage adds 18 representative numerical cases, reusing the existing tiny GEMM and fused-op parity suites plus a focused CUDA router test. The collective test covers both native unicast and multicast, changing-input graph replay, interleaved regular all-reduce, and both TP8 launch shapes. These checks complement the model E2E test from #45257.

The optional SM103 single-token BF16 MoE GEMV and shared SiLU helper moved to draft #46401, with their own correctness test and performance benchmark. They are outside Day-0 kernel support; performance measurements for that follow-up remain pending.

This part changes 21 files (+1,364 / -34).

+/// Every thread holds eight adjacent BF16 values of one row.
+template <uint32_t Hidden, uint32_t Cluster>
+SGL_DEVICE float inv_rms(const Vec& v, float eps) {
+  using namespace device;
+  constexpr uint32_t warps = Hidden / 8 / Cluster / 32;
+  __shared__ float partial[Cluster][warps];
+  float sum = 0.0f;
+#pragma unroll
+  for (uint32_t j = 0; j < 4; ++j) {
+    const auto f = cast<fp32x2_t>(v[j]);
+    sum += f.x * f.x;
+    sum += f.y * f.y;
+  }
+  sum = warp::reduce_sum(sum);
+  const auto lane = threadIdx.x % 32;
+  if constexpr (Cluster > 1) {
+    const auto cluster = cooperative_groups::this_cluster();
+    if (lane < Cluster) *cluster.map_shared_rank(&partial[blockIdx.y][threadIdx.x / 32], lane) = sum;
+    cluster.sync();
+  } else {
+    if (lane == 0) partial[0][threadIdx.x / 32] = sum;
+    __syncthreads();
+  }
+  sum = lane < Cluster * warps ? partial[lane / warps][lane % warps] : 0.0f;
+  sum = warp::reduce_sum(sum);
+  if constexpr (Cluster > 1)
+    cooperative_groups::this_cluster().sync();
+  else
+    __syncthreads();
+  return rsqrtf(sum / float(Hidden) + eps);
+}

+/// Validate and launch a BF16 all-reduce + sandwich/residual/next norm.
+template <uint32_t World, uint32_t Hidden, uint32_t Cluster, uint32_t TopK>
+struct BeamAllReduceNorm {
+  static_assert(World == 2 || World == 4 || World == 8);
+  // Whole warps per CTA, and at most 32 in a cluster so inv_rms reduces them in one warp.
+  static_assert(Hidden % (Cluster * 256) == 0 && Hidden <= 8192);
+  static void
+  run(host::distributed::CommunicatorRef ref,
+      tvm::ffi::TensorView input,
+      tvm::ffi::TensorView residual,
+      tvm::ffi::TensorView post,
+      std::optional<tvm::ffi::TensorView> next,
+      tvm::ffi::TensorView combined,
+      std::optional<tvm::ffi::TensorView> normalized,
+      double eps,
+      double x_scale,
+      double residual_scale,
+      double routed_scale) {
+    using namespace host;
+    const auto& push = ref.get()->get_push_obj();
+    CHECK_HOST(push.world_size == World);
+    SymbolicSize rows{"rows"};
+    SymbolicDevice device;
+    const auto matrix = [&](tvm::ffi::TensorView tensor) {
+      TensorMatcher({rows, Hidden})
+          .with_strides({Hidden, 1})
+          .with_dtype<bf16_t>()
+          .with_device<kDLCUDA>(device)
+          .ensure_alignment(16)
+          .verify(tensor);
+    };
+    const auto weight = [&](tvm::ffi::TensorView tensor) {
+      TensorMatcher({Hidden})
+          .with_strides({1})
+          .with_dtype<bf16_t>()
+          .with_device<kDLCUDA>(device)
+          .ensure_alignment(16)
+          .verify(tensor);
+    };
+    if constexpr (TopK == 1)
+      matrix(input);
+    else
+      TensorMatcher({rows, TopK, Hidden})
+          .with_strides({TopK * Hidden, Hidden, 1})
+          .with_dtype<bf16_t>()
+          .with_device<kDLCUDA>(device)
+          .ensure_alignment(16)
+          .verify(input);
+    matrix(residual);
+    matrix(combined);
+    weight(post);
+    CHECK_HOST(next.has_value() == normalized.has_value());
+    if (next) {
+      weight(*next);
+      matrix(*normalized);
+    }
+    const auto n = rows.unwrap();
+    const int64_t bytes = n * Hidden * sizeof(bf16_t);
+    CHECK_HOST(n > 0 && n <= push.num_blocks && bytes <= push.slot_bytes);
+    const auto p = beam_ar::Params<World>{
+        .input = static_cast<const bf16_t*>(input.data_ptr()),
+        .residual = static_cast<const bf16_t*>(residual.data_ptr()),
+        .post_weight = static_cast<const bf16_t*>(post.data_ptr()),
+        .next_weight = next ? static_cast<const bf16_t*>(next->data_ptr()) : nullptr,
+        .combined = static_cast<bf16_t*>(combined.data_ptr()),
+        .normalized = normalized ? static_cast<bf16_t*>(normalized->data_ptr()) : nullptr,
+        .eps = static_cast<float>(eps),
+        .x_scale = static_cast<float>(x_scale),
+        .residual_scale = static_cast<float>(residual_scale),
+        .routed_scale = static_cast<float>(routed_scale),
+        .rows = static_cast<uint32_t>(n),
+        .rank = push.rank,
+        .counters = push.num_blocks,
+        .workspace = push.get_workspace<World>(bytes)};
+    auto kernel = next ? beam_ar::kernel<World, Hidden, Cluster, TopK, true>
+                       : beam_ar::kernel<World, Hidden, Cluster, TopK, false>;
+    LaunchKernel(dim3(n + (n < push.num_blocks), Cluster), Hidden / 8 / Cluster, input.device())
+        .enable_pdl(true)(kernel, p);
+  }
+};
+}  // namespace sglang

+@triton.jit
+def _fused_attention_gate_kernel(
+    out_ptr,
+    attn_ptr,
+    gate_ptr,
+    bias_ptr,
+    hidden_dim: tl.constexpr,
+    gate_row_stride: tl.constexpr,
+    gate_col_stride: tl.constexpr,
+    HEAD_DIM: tl.constexpr,
+    BLOCK_H: tl.constexpr,
+):
+    row = tl.program_id(0).to(tl.int64)
+    block = tl.program_id(1)
+    offsets = block * BLOCK_H + tl.arange(0, BLOCK_H)
+    mask = offsets < hidden_dim
+    head = offsets // HEAD_DIM
+    act_dtype = attn_ptr.dtype.element_ty
+
+    off = row * hidden_dim + offsets
+    attn = tl.load(attn_ptr + off, mask=mask, other=0.0).to(tl.float32)
+    gate_off = row * gate_row_stride + offsets * gate_col_stride
+    g = tl.load(gate_ptr + gate_off, mask=mask, other=0.0).to(tl.float32)
+    b = tl.load(bias_ptr + head, mask=mask, other=0.0).to(tl.float32)
+    # As in eager, gate + bias, its sigmoid, and the product round to act_dtype.
+    g = (g + b).to(act_dtype).to(tl.float32)

+        # Approximate exp/div can cross an FP16 rounding boundary near zero.
+        m = tl.div_rn(1.0, 1.0 + libdevice.exp(-g))
+    else:
+        m = tl.sigmoid(g)
+    m = m.to(act_dtype).to(tl.float32)
+    tl.store(out_ptr + off, (attn * m).to(act_dtype), mask=mask)
