## Purpose

MRV2 did not propagate `SchedulerOutput.num_spec_tokens_to_schedule` to autoregressive speculators, causing unnecessary draft forwards and stale draft tokens.

This PR fixes the issue

## Test

Acc covered in current unit test

from types import SimpleNamespace

import torch
import triton

from vllm.v1.spec_decode.dynamic.utils import build_dynamic_sd_schedule_lookup
from vllm.v1.worker.gpu.input_batch import InputBatch, InputBuffers

        )

        self.hidden_states = torch.zeros(
            MAX_BATCH_SIZE, HIDDEN_SIZE, dtype=torch.bfloat16, device=device
        )
        self.last_token_indices = torch.zeros(
            MAX_BATCH_SIZE, dtype=torch.int64, device=device
        )
        self.current_draft_step = torch.zeros((), dtype=torch.int64, device=device)
        self.sample_src_positions = torch.zeros(

        self.seeds = torch.zeros(MAX_BATCH_SIZE, dtype=torch.int64, device=device)

            MAX_BATCH_SIZE, MAX_K, dtype=torch.int64, device=device
        )

        self.weight = torch.randn(
            HIDDEN_SIZE, HIDDEN_SIZE, dtype=torch.bfloat16, device=device
        )
        self.src = torch.randn(
            MAX_BATCH_SIZE, HIDDEN_SIZE, dtype=torch.bfloat16, device=device
        )
        self.dst = torch.empty_like(self.src)
        self.count_steps = False
        self.steps = 0

    def _synthetic_forward(self, num_tokens: int) -> None:
        torch.mm(self.src[:num_tokens], self.weight, out=self.dst[:num_tokens])
        self.src, self.dst = self.dst, self.src
        if self.count_steps:
            self.steps += 1

    def _prefill(self, num_reqs: int, num_tokens: int, *args, **kwargs) -> None:
        self._synthetic_forward(num_tokens)
        self.draft_tokens[:num_reqs, 0].zero_()

    def _generate_draft(
        self, num_reqs: int, num_tokens_padded: int, *args, **kwargs
    ) -> None:
        self._synthetic_forward(num_tokens_padded)


def count_forwards(speculator: SyntheticSpeculator, run) -> int:
    speculator.steps = 0
    speculator.count_steps = True
    run()
    speculator.count_steps = False
    return speculator.steps


def main() -> None:
    if not torch.cuda.is_available():
        raise RuntimeError("CUDA is required")

    device = torch.device("cuda")
    speculator = SyntheticSpeculator(device)

    target_hidden_states = torch.randn_like(speculator.hidden_states)
    temperature = torch.zeros(MAX_BATCH_SIZE, device=device)

        DYNAMIC_K_SCHEDULE,
        vllm_max_batch_size=MAX_BATCH_SIZE,
        vllm_num_speculative_tokens=MAX_K,
    )

    print("batch  sched_k  buggy_ms  fixed_ms  speedup  forwards(buggy->fixed)")
    for num_reqs in NUM_REQS:
        batch = InputBatch.make_dummy(num_reqs, num_reqs, target_buffers)
        common_args = dict(
            input_batch=batch,
            attn_metadata={},
            slot_mappings={},
            last_hidden_states=target_hidden_states[:num_reqs],
            aux_hidden_states=None,
            num_sampled=torch.ones(num_reqs, dtype=torch.int32, device=device),
            num_rejected=torch.zeros(num_reqs, dtype=torch.int32, device=device),
            last_sampled=last_sampled,
            next_prefill_tokens=target_buffers.input_ids,
            temperature=temperature,
            seeds=seeds,
            dummy_run=True,
            skip_attn_for_dummy_run=True,
            is_profile=True,
        )
        dynamic_k = dense_schedule[num_reqs]

        # Buggy: model_runner omitted runtime K, so propose() used max K=5.
        buggy = partial(speculator.propose, **common_args)
        # Fixed: model_runner passes the scheduler-selected runtime K=3.
        fixed = partial(
            speculator.propose,
            **common_args,
            num_speculative_tokens=dynamic_k,
        )

        buggy_steps = count_forwards(speculator, buggy)
        fixed_steps = count_forwards(speculator, fixed)
        assert dynamic_k == 3
        assert buggy_steps == MAX_K
        assert fixed_steps == dynamic_k

        buggy_ms = triton.testing.do_bench(buggy)
        fixed_ms = triton.testing.do_bench(fixed)
        print(
            f"{num_reqs:>5}  {dynamic_k:>7}  {buggy_ms:>8.3f}  "
            f"{fixed_ms:>8.3f}  {buggy_ms / fixed_ms:>7.3f}x  "
            f"{buggy_steps}->{fixed_steps}"
        )


if __name__ == "__main__":
    main()
