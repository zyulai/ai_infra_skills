```markdown
**Motivation.** Under attention DP, every Kimi-K3 PD prefill rank holds every KDA layer's `[q,k,v,g]` in-projection
and `o_proj` (60.8 GB BF16 per rank at attention TP 1), plus 24 GB of replicated shared experts under an EP
all-to-all backend. ADP8 then barely fits on 8 x GB300.

**Change.** `--kda-dwdp-size N` (DWDP for the KDA projection and shared-expert weights; opt-in, `--disaggregation-mode prefill` only) shards those weights across N
attention-DP ranks, and gathers the next layer into the other of two rotating slots on a copy stream while the current layer computes.
- **Head-aligned layout:** rank r stores exactly tensor-parallel rank r's slice (heads/N rows of each of q / k / v /
  g, the matching o_proj columns), so with N == TP the shard is byte-identical to KDA TP rank r's weights. N may be
  any divisor of the attention-DP size that splits the heads.
- **Gather:** copy-engine peer pulls from a cuMem shard mapped on every group rank with `cuda_vmm_utils` (one copy per in-proj block, one
  `cuMemcpy2DAsync` per o_proj column slice). It uses no SMs, no collective and no replica. The slot is the
  replicated layout, so prefill outputs are bitwise identical to replicated weights.
- **Gates:** prefill role only; refuses PP, attention CP, full CUDA graphs, loaders that skip `load_weights`, the weight
  cache, quantized sharded weights, a second K3 model, online weight updates, and non-Kimi-K3 models.

+
+from __future__ import annotations
+
+from typing import List, Sequence, Tuple
+
+import msgspec
+
+# 256 B for bf16, so parameter views keep the alignment cuBLAS and TMA expect.
+ALIGN = 128
+
+
+def _align(numel: int) -> int:
+    return numel + (-numel) % ALIGN
+
+
+class ParamSpec(msgspec.Struct, frozen=True):
+    shape: Tuple[int, int]
+    slot_offset: int
+    # > 0: each of this many row blocks is cut N ways; 0: the columns are cut N ways.
+    row_blocks: int
+
+
+class Piece(msgspec.Struct, frozen=True):
+    """This rank's part of one row block, or its column slice, of a parameter."""
+
+    param: int
+    block: int
+    shard_offset: int
+    rows: int
+    cols: int
+
+
+class CopyOp(msgspec.Struct, frozen=True):
+    src_rank: int
+    src_offset: int
+    dst_offset: int
+    rows: int
+    cols: int
+    # Elements between destination rows; 0 for a contiguous copy.
+    dst_pitch: int
+
+
+class UnitLayout:
+    """One layer's parameters, as ``(shape, row_blocks)``, split over ``group_size`` ranks."""
+
+    def __init__(self, params: Sequence[Tuple[Tuple[int, int], int]], group_size: int):
+        self.group_size = group_size
+        self.specs: List[ParamSpec] = []
+        self.slot_numel = 0
+        for shape, row_blocks in params:
+            self.specs.append(ParamSpec(tuple(shape), self.slot_numel, row_blocks))
+            self.slot_numel = _align(self.slot_numel + shape[0] * shape[1])
+        self.pieces: List[Piece] = []
+        self.shard_numel = 0
+        for i, spec in enumerate(self.specs):
+            rows, cols = spec.shape
+            if spec.row_blocks:
+                if rows % (spec.row_blocks * group_size):
+                    raise ValueError(
+                        f"{rows} rows do not split into {spec.row_blocks} blocks x "
+                        f"{group_size} ranks"
+                    )
+                dims = [
+                    (b, rows // (spec.row_blocks * group_size), cols)
+                    for b in range(spec.row_blocks)
+                ]
+            else:
+                if cols % group_size:
+                    raise ValueError(
+                        f"{cols} columns do not split over {group_size} ranks"
+                    )
+                dims = [(0, rows, cols // group_size)]
+            for block, piece_rows, piece_cols in dims:
+                self.pieces.append(
+                    Piece(i, block, self.shard_numel, piece_rows, piece_cols)
+                )
+                self.shard_numel = _align(self.shard_numel + piece_rows * piece_cols)
+
+    def first_row(self, piece: Piece, rank: int) -> int:
+        """The parameter row where ``rank``'s copy of a row-block piece starts."""
+        return (piece.block * self.group_size + rank) * piece.rows
+
+    def copy_ops(self, rank: int) -> List[CopyOp]:
+        n = self.group_size
+        ops = []
+        # Rotated peer order so no source serves two readers at once; own pieces last.
+        for src in [(rank + i) % n for i in range(1, n)] + [rank]:
+            for pc in self.pieces:
+                spec = self.specs[pc.param]
+                if spec.row_blocks:
+                    dst, pitch = spec.slot_offset + self.first_row(pc, src) * pc.cols, 0
+                else:
+                    dst, pitch = spec.slot_offset + src * pc.cols, spec.shape[1]
+                ops.append(CopyOp(src, pc.shard_offset, dst, pc.rows, pc.cols, pitch))
+        return ops
+

+"""Double-buffered prefetch of per-layer weights on a copy stream."""
+
+from __future__ import annotations
+
+from typing import Callable, Dict, List
+
+import torch
+
+
+class SlotPrefetcher:
+    """Copies unit ``u`` into slot ``u % num_slots`` on a copy stream, ahead of the compute that reads it."""
+
+    def __init__(
+        self,
+        copy_unit: Callable[[int, torch.cuda.Stream], None],
+        *,
+        num_slots: int,
+        device: torch.device,
+    ):
+        self._copy_unit = copy_unit
+        self.num_slots = num_slots
+        self._device = device
+        self._stream = torch.cuda.Stream(device=device)
+        self._stream_ctx = torch.cuda.stream(self._stream)
+        self._ready = [torch.cuda.Event() for _ in range(num_slots)]
+        self._consumed = [torch.cuda.Event() for _ in range(num_slots)]
+        current = torch.cuda.current_stream(device)
+        for ev in self._consumed:
+            ev.record(current)
+        # unit -> ready event, for copies issued and not yet waited on.
+        self._pending: Dict[int, torch.cuda.Event] = {}
+        self._issued: List[int] = []
+
+    def issue(self, unit: int) -> None:
+        if unit not in self._pending:
+            self._issue(unit)
+            self._issued.append(unit)
+
+    def _issue(self, unit: int) -> None:
+        if torch.cuda.is_current_stream_capturing():
+            raise RuntimeError(
+                "DWDP weight copies must be issued outside CUDA graph capture: run "
+                "prefill eager or with the breakable prefill graph (the hooks are eager "
+                "breaks), and disable decode / full graphs"
+            )
+        slot = unit % self.num_slots
+        # Compute has finished reading the slot's previous unit.
+        self._stream.wait_event(self._consumed[slot])
+        with self._stream_ctx:
+            self._copy_unit(unit, self._stream)
+            self._ready[slot].record(self._stream)
+        self._pending[unit] = self._ready[slot]
+
+    def wait(self, unit: int) -> None:
+        if unit not in self._pending:
+            self._issue(unit)
+        torch.cuda.current_stream(self._device).wait_event(self._pending.pop(unit))
+
+    def release(self, unit: int) -> None:
+        self._consumed[unit % self.num_slots].record(
+            torch.cuda.current_stream(self._device)
+        )
+
+    def wait_issued(self) -> None:
+        issued, self._issued = self._issued, []
+        current = torch.cuda.current_stream(self._device)
+        for unit in issued:
+            ev = self._pending.get(unit)
+            if ev is not None:
+                current.wait_event(ev)
